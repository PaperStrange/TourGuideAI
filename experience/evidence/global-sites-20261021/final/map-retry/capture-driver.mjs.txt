import { chromium } from '../experience/node_modules/playwright/index.mjs';
import { createServer } from 'node:http';
import { readFileSync, readdirSync, mkdirSync, writeFileSync } from 'node:fs';
import { resolve, join, relative, extname } from 'node:path';
import { createHash } from 'node:crypto';

const root = resolve(import.meta.dirname, '..');
const output = resolve(root, process.env.MAP_RETRY_OUTPUT || 'artifacts/style-retry');
const dist = resolve(root, 'experience/dist');
mkdirSync(output, { recursive: true });
process.env.TMPDIR = join(output, 'tmp'); mkdirSync(process.env.TMPDIR, { recursive: true });
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const assets = new Map();
function collect(directory) {
  for (const entry of readdirSync(directory, { withFileTypes: true })) {
    const file = join(directory, entry.name);
    if (entry.isDirectory()) collect(file);
    else assets.set('/' + relative(dist, file), readFileSync(file));
  }
}
collect(dist);
const previous = JSON.parse(readFileSync(join(root, 'experience/evidence/global-sites-20261021/ci/solo-result.json')));
const saved = previous.checks.find(check => check.id === 'map-save-keeps-center-orientation-and-selected-order').evidence.saved;
const key = 'tourguideai.solo.places.v1';
const seed = JSON.stringify({ version: 1, places: [saved] });
const files = [...assets].sort(([a], [b]) => a.localeCompare(b)).map(([path, bytes]) => ({ path, bytes: bytes.length, sha256: sha(bytes) }));
const report = { startedAt: new Date().toISOString(), driverSha256: sha(readFileSync(new URL(import.meta.url))),
  artifact: { files, manifestSha256: sha(JSON.stringify(files)) }, scenarios: [],
  conditions: { viewport: [1440, 1000], reducedMotion: 'reduce', tlsVerification: true,
    source: 'Actual saved map view from retained CI report; initial request denial and real provider retry.' } };
const persist = () => writeFileSync(join(output, 'result.json'), JSON.stringify(report, null, 2) + '\n');
const mime = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.json': 'application/json',
  '.png': 'image/png', '.jpg': 'image/jpeg', '.svg': 'image/svg+xml', '.glb': 'model/gltf-binary', '.woff': 'font/woff' };
const server = createServer((request, response) => {
  const path = new URL(request.url, 'http://localhost').pathname;
  const bytes = assets.get(path === '/' ? '/index.html' : path);
  response.writeHead(bytes ? 200 : 404, { 'Content-Type': mime[extname(path === '/' ? '/index.html' : path)] || 'application/octet-stream' });
  response.end(bytes || 'Unavailable');
});
await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
const origin = `http://127.0.0.1:${server.address().port}`;
const proxyValue = process.env.HTTPS_PROXY || process.env.HTTP_PROXY;
const proxy = proxyValue ? new URL(proxyValue) : null;
let browser;
try {
  browser = await chromium.launch({ executablePath: '/usr/bin/chromium', headless: true, args: ['--no-sandbox'],
    ...(proxy ? { proxy: { server: proxy.origin, bypass: 'localhost,127.0.0.1', ...(proxy.username ? {
      username: decodeURIComponent(proxy.username), password: decodeURIComponent(proxy.password) } : {}) } } : {}) });
  report.conditions.browser = browser.version();
  const scenarios = process.env.MAP_RETRY_BASELINE ? ['style-en'] : ['style-en', 'style-zh', 'tiles-en'];
  for (const name of scenarios) {
    const locale = name.endsWith('zh') ? 'zh' : 'en';
    const result = { name, styles: [], denied: [], errors: [], responses: [], checks: [] };
    report.scenarios.push(result);
    const check = (id, pass, evidence) => { result.checks.push({ id, pass: Boolean(pass), evidence }); persist(); console.log(`${pass ? 'PASS' : 'FAIL'} ${name}: ${id}`); };
    const context = await browser.newContext({ viewport: { width: 1440, height: 1000 }, deviceScaleFactor: 1, reducedMotion: 'reduce' });
    await context.addInitScript(({ key, seed }) => localStorage.setItem(key, seed), { key, seed });
    let deny = true;
    await context.route('https://tiles.openfreemap.org/**', route => {
      const url = route.request().url();
      if (url.endsWith('/styles/liberty')) result.styles.push({ url, time: Date.now() });
      const selected = name.startsWith('style') ? url.endsWith('/styles/liberty') : /\.(pbf|mvt)(\?|$)/.test(url);
      if (deny && selected) { result.denied.push(url); return route.abort('blockedbyclient'); }
      return route.continue();
    });
    const page = await context.newPage(); page.setDefaultTimeout(90000);
    page.on('pageerror', error => result.errors.push(error.message));
    page.on('response', response => { if (response.url().startsWith('https://tiles.openfreemap.org/')) result.responses.push({ url: response.url(), status: response.status() }); });
    const snapshot = () => page.evaluate(key => ({ map: window.__HOME__?.snapshot().map,
      status: document.querySelector('#scene-status').textContent, retryVisible: !document.querySelector('#scene-retry').hidden,
      soloStatus: document.querySelector('#solo-status').textContent, view: window.__HOME__?.snapshot().currentView,
      saved: localStorage.getItem(key), markers: [...document.querySelectorAll('.destination-pin')].map(node => node.textContent) }), key);
    try {
      await page.goto(`${origin}/?qa=1&lang=${locale}`, { waitUntil: 'domcontentloaded' });
      await page.locator(`[data-kept-open="${saved.id}"]`).click();
      await page.waitForFunction(() => /could not fully reopen|未能完整/.test(document.querySelector('#solo-status').textContent));
      result.before = await snapshot();
      check('failure-visible-and-save-preserved', result.denied.length && result.before.retryVisible && result.before.map.failed && result.before.saved === seed, result.before);
      deny = false;
      const count = result.styles.length;
      await page.locator('#scene-retry').click();
      if (process.env.MAP_RETRY_BASELINE) {
        await page.waitForTimeout(5000); result.after = await snapshot();
        check('retry-dispatches-new-style', result.styles.length > count, { before: count, after: result.styles.length, state: result.after });
      } else {
        await page.waitForFunction(() => window.__HOME__?.snapshot().map?.ready);
        result.after = await snapshot();
        check('retry-dispatches-new-style', result.styles.length > count, result.styles);
        check('provider-ready-with-markers-and-no-exceptions', result.after.map.ready && !result.after.map.failed && result.after.markers.length === saved.route.stops.length && result.errors.length === 0, result.after);
        check('retry-preserves-camera-places-routes-and-save', ['center', 'zoom', 'pitch', 'bearing', 'places', 'routes', 'cameraTransitions'].every(key => JSON.stringify(result.before.map[key]) === JSON.stringify(result.after.map[key])) && result.after.saved === seed, result.after);
        await page.locator(`[data-kept-open="${saved.id}"]`).click();
        await page.waitForFunction(() => /saved view and place order are open|已打开收藏/.test(document.querySelector('#solo-status').textContent));
        const restored = await snapshot();
        check('saved-map-restores-exact-pose', JSON.stringify(restored.view) === JSON.stringify(saved.view) && restored.saved === seed, restored);
        await page.locator(locale === 'en' ? '#locale-zh' : '#locale-en').click();
        const translated = await snapshot();
        check('locale-keeps-ready-camera-and-markers', translated.map.ready && JSON.stringify(translated.view) === JSON.stringify(saved.view) && translated.markers.length === saved.route.stops.length, translated);
      }
    } catch (error) { check('scenario-completed', false, { message: error.message, stack: error.stack, state: await snapshot() }); }
    finally { await context.close(); }
  }
  report.passed = report.scenarios.every(scenario => scenario.checks.every(check => check.pass));
  if (!report.passed) process.exitCode = 1;
} finally {
  await browser?.close(); server.closeAllConnections(); await new Promise(resolve => server.close(resolve));
  report.finishedAt = new Date().toISOString(); persist();
}
